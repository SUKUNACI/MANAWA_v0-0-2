QUICK COMMANDS
================================================================================

# Verify Environment Variables
echo $CLOUDINARY_CLOUD_NAME $CLOUDINARY_API_KEY

# Test CDN Health
curl http://localhost:3000/admin/cdn/health

# Get Usage Stats
curl -H "Authorization: Bearer TOKEN" \
  http://localhost:3000/admin/cdn/stats

# List Resources
curl -H "Authorization: Bearer TOKEN" \
  "http://localhost:3000/admin/cdn/resources?folder=manawa/avatars"

# Purge Cache by Folder
curl -X POST \
  -H "Authorization: Bearer TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"method": "folder", "folder": "manawa/avatars"}' \
  http://localhost:3000/admin/cdn/purge

# Cleanup Old Images
curl -X POST \
  -H "Authorization: Bearer TOKEN" \
  "http://localhost:3000/admin/cdn/cleanup?daysOld=30"
